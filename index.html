/** * 학생 알림장 도우미 - 백엔드 (Google Apps Script) * 시트 구성 * - 학생DB : 이름 | 학교학년 | 시험범위 | 시험날짜 | 교과서 | 시험기간 * (시험기간 열이 없다면 F열에 헤더를 "시험기간"으로 추가해주세요 — 전체 시험기간을 담습니다. 시험날짜는 과학시험날짜로 사용합니다) * - 과제DB : 교재명 | 진도단원 | 과제 | 공통알림장 * - 수업기록 : 이 스크립트가 자동 생성/관리 (기존 데이터가 있어도 헤더 이름으로 열을 찾으므로 안전합니다) * * 배포 방법: 배포 > 배포 관리 > 편집(연필) > 새 버전으로 배포 * (기존 배포 URL을 그대로 쓰려면 "새 배포"가 아니라 "편집 > 새 버전"으로 올려야 URL이 안 바뀝니다) */ function doGet(e) { try { const action = (e.parameter.action || 'getData'); if (action === 'getData') { return respond(getAllData()); } return respond({ error: '알 수 없는 action: ' + action }); } catch (err) { return respond({ error: err.toString() }); } } function doPost(e) { try { const action = e.parameter.action || 'addRecord'; let result; switch (action) { case 'addStudent': result = addStudent(e.parameter); break; case 'updateStudent': result = updateStudent(e.parameter); break; case 'addCurriculumBatch': result = addCurriculumBatch(JSON.parse(e.parameter.items || '[]')); break; case 'updateCurriculum': result = updateCurriculum(e.parameter); break; case 'deleteRecord': result = deleteRecord(e.parameter); break; case 'addRecord': default: result = addRecord(e.parameter); break; } return respond(result); } catch (err) { return respond({ error: err.toString() }); } } function respond(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } function getSS() { return SpreadsheetApp.getActiveSpreadsheet(); } function getHeaderIndex(headers, names) { for (const n of names) { const idx = headers.indexOf(n); if (idx !== -1) return idx; } return -1; } function getStudentSheet(ss) { return ss.getSheetByName('학생DB') || ss.getSheetByName('학생목록') || ss.getSheets()[0]; } function getCurriSheet(ss) { return ss.getSheetByName('과제DB') || ss.getSheetByName('커리큘럼') || ss.getSheetByName('진도표'); } function getAllData() { const ss = getSS(); // 1. 학생 목록 let students = []; const studentSheet = getStudentSheet(ss); if (studentSheet) { const data = studentSheet.getDataRange().getValues(); if (data.length > 1) { const headers = data[0]; const nameIdx = getHeaderIndex(headers, ['이름']); const schoolIdx = getHeaderIndex(headers, ['학교학년']); const rangeIdx = getHeaderIndex(headers, ['시험범위']); const dateIdx = getHeaderIndex(headers, ['시험날짜']); const periodIdx = getHeaderIndex(headers, ['시험기간']); const bookIdx = getHeaderIndex(headers, ['교과서']); for (let i = 1; i < data.length; i++) { const row = data[i]; const name = nameIdx !== -1 ? row[nameIdx] : row[0]; if (name && name.toString().trim() !== '') { students.push({ 이름: name.toString().trim(), 학교학년: schoolIdx !== -1 && row[schoolIdx] ? row[schoolIdx].toString().trim() : '', 시험범위: rangeIdx !== -1 && row[rangeIdx] ? row[rangeIdx].toString().trim() : '', 시험날짜: dateIdx !== -1 && row[dateIdx] ? formatDate(row[dateIdx]) : '', 시험기간: periodIdx !== -1 && row[periodIdx] ? row[periodIdx].toString().trim() : '', 교과서: bookIdx !== -1 && row[bookIdx] ? row[bookIdx].toString().trim() : '' }); } } } } // 2. 커리큘럼(교재/진도/과제/공통알림장) let curriculum = {}; const curriSheet = getCurriSheet(ss); if (curriSheet) { const data = curriSheet.getDataRange().getValues(); if (data.length > 1) { let currentBook = ''; for (let i = 1; i < data.length; i++) { const row = data[i]; let book = row[0] ? row[0].toString().trim() : ''; let lesson = row[1] ? row[1].toString().trim() : ''; let homework = row[2] ? row[2].toString().trim() : ''; let commonNotice = row[3] ? row[3].toString().trim() : ''; if (book !== '') currentBook = book; else book = currentBook; if (book && lesson) { if (!curriculum[book]) curriculum[book] = []; curriculum[book].push({ lesson: lesson, homework: homework, commonNotice: commonNotice }); } } } } // 3. 수업기록 (최근 과제/시험점수/결석·보강 조회용) let records = []; const recordSheet = ss.getSheetByName('수업기록'); if (recordSheet) { const data = recordSheet.getDataRange().getValues(); if (data.length > 1) { const headers = data[0]; for (let i = 1; i < data.length; i++) { const row = data[i]; let obj = {}; headers.forEach((h, idx) => { obj[h] = row[idx] instanceof Date ? formatDate(row[idx]) : row[idx]; }); records.push(obj); } } } return { students: students, curriculum: curriculum, records: records }; } function addStudent(p) { const ss = getSS(); const sheet = getStudentSheet(ss); if (!sheet) return { error: '학생DB 시트를 찾을 수 없습니다' }; const data = sheet.getDataRange().getValues(); const headers = data[0]; const nameIdx = getHeaderIndex(headers, ['이름']); const name = (p.name || '').toString().trim(); if (!name) return { error: '이름이 없습니다' }; for (let i = 1; i < data.length; i++) { if (data[i][nameIdx] && data[i][nameIdx].toString().trim() === name) { return { error: '이미 존재하는 학생입니다', duplicate: true }; } } const row = new Array(headers.length).fill(''); const set = (names, val) => { const idx = getHeaderIndex(headers, names); if (idx !== -1) row[idx] = val || ''; }; set(['이름'], name); set(['학교학년'], p.schoolGrade); set(['시험범위'], p.examRange); set(['시험날짜'], p.examDate); set(['시험기간'], p.examPeriod); set(['교과서'], p.textbook); sheet.appendRow(row); return { result: 'success' }; } function updateStudent(p) { const ss = getSS(); const sheet = getStudentSheet(ss); if (!sheet) return { error: '학생DB 시트를 찾을 수 없습니다' }; const data = sheet.getDataRange().getValues(); const headers = data[0]; const nameIdx = getHeaderIndex(headers, ['이름']); const name = (p.name || '').toString().trim(); if (!name) return { error: '이름이 없습니다' }; for (let i = 1; i < data.length; i++) { if (data[i][nameIdx] && data[i][nameIdx].toString().trim() === name) { const rowNum = i + 1; const setCell = (names, val) => { if (val === undefined || val === null || val === '') return; const idx = getHeaderIndex(headers, names); if (idx !== -1) sheet.getRange(rowNum, idx + 1).setValue(val); }; setCell(['학교학년'], p.schoolGrade); setCell(['시험범위'], p.examRange); setCell(['시험날짜'], p.examDate); setCell(['시험기간'], p.examPeriod); setCell(['교과서'], p.textbook); return { result: 'success' }; } } return { error: '학생을 찾을 수 없습니다' }; } function addCurriculumBatch(items) { const ss = getSS(); const sheet = getCurriSheet(ss); if (!sheet) return { error: '과제DB 시트를 찾을 수 없습니다' }; const data = sheet.getDataRange().getValues(); let currentBook = ''; const existingKeys = new Set(); for (let i = 1; i < data.length; i++) { let book = data[i][0] ? data[i][0].toString().trim() : ''; let lesson = data[i][1] ? data[i][1].toString().trim() : ''; if (book !== '') currentBook = book; else book = currentBook; if (book && lesson) existingKeys.add(book + '|||' + lesson); } let added = 0; (items || []).forEach(item => { const book = (item.book || '').toString().trim(); const lesson = (item.lesson || '').toString().trim(); if (!book || !lesson) return; const key = book + '|||' + lesson; if (existingKeys.has(key)) return; sheet.appendRow([book, lesson, item.homework || '', item.commonNotice || '']); existingKeys.add(key); added++; }); return { result: 'success', added: added, skipped: (items || []).length - added }; } function updateCurriculum(p) { const ss = getSS(); const sheet = getCurriSheet(ss); if (!sheet) return { error: '과제DB 시트를 찾을 수 없습니다' }; const book = (p.book || '').toString().trim(); const lesson = (p.lesson || '').toString().trim(); if (!book || !lesson) return { error: '교재명과 진도단원이 필요합니다' }; const data = sheet.getDataRange().getValues(); let currentBook = ''; for (let i = 1; i < data.length; i++) { let rowBook = data[i][0] ? data[i][0].toString().trim() : ''; let rowLesson = data[i][1] ? data[i][1].toString().trim() : ''; if (rowBook !== '') currentBook = rowBook; else rowBook = currentBook; if (rowBook === book && rowLesson === lesson) { const rowNum = i + 1; if (p.newLesson !== undefined && p.newLesson !== '') sheet.getRange(rowNum, 2).setValue(p.newLesson); if (p.homework !== undefined) sheet.getRange(rowNum, 3).setValue(p.homework); if (p.commonNotice !== undefined) sheet.getRange(rowNum, 4).setValue(p.commonNotice); return { result: 'success' }; } } return { error: '해당 교재/진도를 찾을 수 없습니다' }; } function addRecord(p) { const ss = getSS(); let recordSheet = ss.getSheetByName('수업기록'); const wantedHeaders = ['작성일시', '학생이름', '교재명', '오늘진도', '오늘과제', '개념암기결과', '객관식점수', '결석일자', '보강일자', '보강시간', '선생님전달사항']; if (!recordSheet) { recordSheet = ss.insertSheet('수업기록'); recordSheet.appendRow(wantedHeaders); } const headers = recordSheet.getDataRange().getValues()[0]; const now = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd HH:mm:ss'); const row = new Array(headers.length).fill(''); const set = (names, val) => { const idx = getHeaderIndex(headers, names); if (idx !== -1) row[idx] = val || ''; }; set(['작성일시'], now); set(['학생이름'], p.studentName); set(['교재명'], p.textbook); set(['오늘진도'], p.progress); set(['오늘과제'], p.homework); set(['개념암기결과'], p.conceptResult); set(['객관식점수'], p.quizScore); set(['결석일자'], p.absentDate); set(['보강일자'], p.makeupDate); set(['보강시간'], p.makeupTime); set(['선생님전달사항'], p.memo); recordSheet.appendRow(row); return { result: 'success' }; } function deleteRecord(p) { const ss = getSS(); const recordSheet = ss.getSheetByName('수업기록'); if (!recordSheet) return { error: '수업기록 시트를 찾을 수 없습니다' }; const data = recordSheet.getDataRange().getValues(); const headers = data[0]; const writtenAtIdx = getHeaderIndex(headers, ['작성일시']); const nameIdx = getHeaderIndex(headers, ['학생이름']); const writtenAt = (p.writtenAt || '').toString().trim(); const studentName = (p.studentName || '').toString().trim(); if (!writtenAt || !studentName) return { error: '삭제할 기록을 특정할 수 없습니다' }; for (let i = 1; i < data.length; i++) { const rowWrittenAt = data[i][writtenAtIdx] instanceof Date ? formatDate(data[i][writtenAtIdx]) : (data[i][writtenAtIdx] || '').toString().trim(); const rowName = (data[i][nameIdx] || '').toString().trim(); if (rowWrittenAt === writtenAt && rowName === studentName) { recordSheet.deleteRow(i + 1); return { result: 'success' }; } } return { error: '해당 기록을 찾을 수 없습니다' }; } function formatDate(dateObj) { if (!dateObj) return ''; if (dateObj instanceof Date) { return Utilities.formatDate(dateObj, Session.getScriptTimeZone(), 'yyyy-MM-dd'); } return dateObj.toString(); }
